Grey-headed Lovebird visual-mutation

CREATE TABLE canus_visual_mutations (
  id INT UNSIGNED NOT NULL AUTO_INCREMENT,
  species VARCHAR(64) NOT NULL DEFAULT 'Agapornis canus',
  series VARCHAR(64) NOT NULL,
  mutation VARCHAR(64) NOT NULL,
  sf VARCHAR(255) NOT NULL,
  df VARCHAR(255) NOT NULL,
  allele VARCHAR(255) NOT NULL,
  genotype VARCHAR(255) NOT NULL,
  genetic_code VARCHAR(64) NOT NULL,
  inheritance_type VARCHAR(80) NOT NULL,
  phenotype TEXT NOT NULL,
  scientific_source VARCHAR(255) NOT NULL,
  verification_status VARCHAR(64) NOT NULL,
  computable TINYINT(1) NOT NULL DEFAULT 1,
  PRIMARY KEY (id),
  UNIQUE KEY uq_species_mutation (species, mutation)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
